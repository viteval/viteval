```ts
import { evaluate, scorers } from 'viteval';
import { classifyTicket } from './support-agent';

evaluate('Support routing', {
  data: [
    { input: 'I was charged twice.', expected: 'billing' },
    { input: 'Reset my password.', expected: 'account' },
    { input: 'Cancel my subscription.', expected: 'billing' },
  ],
  task: ({ input }) => classifyTicket(input),
  scorers: [scorers.exactMatch()],
});
```
