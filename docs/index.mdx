```ts
import { evaluate, scorers } from 'viteval';
import { knowledgeQuestions } from './knowledge.dataset';
import { retrieve, rerank, answer } from './knowledge-agent';

evaluate('Knowledge answers', {
  data: knowledgeQuestions,
  task: async ({ input }) => {
    const documents = await retrieve(input, { limit: 20 });
    const context = await rerank({ query: input, documents, topK: 5 });
    const response = await answer({ question: input, context });
    return response.text;
  },
  scorers: [
    scorers.factual(),
    scorers.answerCorrectness(),
    scorers.answerRelevancy(),
  ],
  threshold: 0.85,
});
```

```ts
import { afterAll } from 'vitest';
import { openai } from '@ai-sdk/openai';
import { s3 } from '@viteval/providers/s3';
import { initializeModel } from 'viteval';
import { initializeProvider } from 'viteval/provider';

initializeModel({ language: openai('gpt-4.1-mini') });

const datasets = s3({
  bucket: 'evaluation-data',
  prefix: 'knowledge/datasets',
  clientOptions: { region: 'us-east-1' },
});

const result = await initializeProvider({ datasets });
if (!result.ok) throw result.error;

afterAll(async () => {
  const closed = await datasets.close();
  if (!closed.ok) throw closed.error;
});
```
