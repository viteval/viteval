# `evaluate()`

import { Field, FieldGroup } from '@theme';

`evaluate()` registers an evaluation with a check for each case. It runs your task,
scores the output, and checks the threshold.

## Import

```ts
import { evaluate } from 'viteval';
```

## Signature

```ts
function evaluate<DATA extends Data, TASK_OUTPUT = InferDataOutput<DATA>>(
  name: string,
  {
    data,
    aggregation = 'mean',
    task,
    scorers,
    threshold = 1.0,
    timeout,
    concurrency,
    retries,
    results,
    skip,
  }: Eval<DATA, TASK_OUTPUT>
): void;
```

  <Field name="options" type="Eval<DATA, TASK_OUTPUT>" required>

          <Field name="expected" type="EXPECTED" required>

          <Field name="name" type="string">

      <Field name="task" type="Task<INPUT, TASK_OUTPUT, EXTRA>" required>
        Synchronous or asynchronous function receiving input and extra case

      </Field>
      <Field
        name="scorers"
        type="Scorer<TASK_OUTPUT, EXPECTED, EXTRA>[]"

        [createScorer()](/api/scoring/create-scorer).
      </Field>
      <Field name="threshold" type="number" defaultValue="1">
        A finite minimum for each case's aggregate score. The case passes when
        its aggregate is greater than or equal to this value. Sum thresholds may
        exceed 1.
      </Field>
      <Field
        name="aggregation"
        type="'mean' | 'median' | 'sum'"
        defaultValue="'mean'"
      >
        Combines each case's scorer values. Null contributes zero. Task or
        scorer exceptions and timeouts fail the case regardless of threshold.
      </Field>
      <Field name="timeout" type="number">
        Positive per-case timeout in milliseconds, including task and scoring.
      </Field>
      <Field name="concurrency" type="number">
        Positive case concurrency limit, bounded by the worker limit.
      </Field>
      <Field name="retries" type="number">
        Additional attempts after a failed case; reruns tasks and scorers.
      </Field>
      <Field name="results" type="ResultProviderName[]">
        Result destinations. Omitted inherits config; an explicit array replaces
        it; an empty array disables persistence.
      </Field>
      <Field name="skip" type="boolean">

      </Field>

        Descriptive text that does not affect scoring or case discovery.

## Related types

```ts
type Data<DATA_ITEM extends DataItem> =
  DATA_ITEM[] | DataGenerator<DATA_ITEM> | Dataset<DataGenerator<DATA_ITEM>>;

type Task<INPUT, OUTPUT, EXTRA extends Extra> = (
  args: TaskArgs<INPUT, EXTRA>
) => Promise<OUTPUT> | OUTPUT;

type TaskArgs<INPUT, EXTRA extends Extra> = EXTRA & { input: INPUT };
```

Each case is a `DataItem`. Additional fields are preserved in `EXTRA`:

```ts
type DataItem<
  INPUT = unknown,
  OUTPUT = unknown,
  EXTRA extends Extra = Extra,
> = EXTRA & {
  name?: string;
  input: INPUT;
  expected: OUTPUT;
};
```

`Extra` holds additional case fields:

```ts
type Extra = Record<string, unknown>;
```

## Return value

Returns `void`. Evaluation outcomes are recorded during execution, not returned
as completed scores or runner objects by this call.

## Worker-local dependencies

`data` can select a registered integration's dataset handle. `task` and `scorers`
also accept factories that resolve dependencies from the worker-local context.
Generation destinations do not select the evaluation's read source:

```ts
evaluate('Reviewed support answers', {
  data: ({ langfuse }) => langfuse.dataset('reviewed-support'),
  task: ({ input }) => answerSupportQuestion(input),
  scorers: [scorers.factual()],
  results: ['json'],
});
```

Import and register the Langfuse integration in config before using its handle.
Storage integrations initialize lazily, fail the run on persistence errors, and
close after persistence. A successful write is not rolled back if another
destination fails.

## Example

```ts
import { evaluate, scorers } from 'viteval';

evaluate('Order extraction', {
  data: [
    { input: 'Please cancel ord-2087.', expected: 'ORD-2087' },
    { input: 'How do I update my address?', expected: '' },
  ],
  task: ({ input }) => input.match(/\bORD-\d+\b/i)?.[0].toUpperCase() ?? '',
  scorers: [scorers.exactMatch()],
  threshold: 1,
  timeout: 60_000,
  retries: 2,
});
```
